cmake_minimum_required(VERSION 3.20)
project(reward-desk-obs VERSION 0.10.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(reward-desk-obs MODULE plugin.cpp dock.cpp)
add_executable(reward-desk-obs-preview WIN32 preview.cpp dock.cpp)
foreach(target reward-desk-obs reward-desk-obs-preview)
  target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network shell32 ole32)
  target_compile_definitions(${target} PRIVATE NOMINMAX REWARD_DESK_VERSION="${PROJECT_VERSION}")
  target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive- /Zc:__cplusplus)
endforeach()
